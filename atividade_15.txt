-- ETAPA 1: CRIAÇÃO DAS TABELAS
--------------------------------------------------------------------------------

CREATE TABLE tb_empregados (
    id_empregado NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome         VARCHAR2(100) NOT NULL,
    cargo        VARCHAR2(50) NOT NULL,
    salario      NUMBER(10,2) NOT NULL,
    status       VARCHAR2(20) DEFAULT 'ATIVO' NOT NULL,
    data_criacao DATE DEFAULT SYSDATE
);

CREATE TABLE tb_auditoria_salarios (
    id_audit         NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_empregado     NUMBER NOT NULL,
    operacao         VARCHAR2(10) NOT NULL,
    salario_anterior NUMBER(10,2),
    salario_novo     NUMBER(10,2),
    usuario_db       VARCHAR2(50) DEFAULT USER,
    data_evento      TIMESTAMP DEFAULT SYSTIMESTAMP
);

--------------------------------------------------------------------------------
-- ETAPA 2: TRIGGER BEFORE ROW-LEVEL
-- VALIDAÇÃO E HIGIENIZAÇÃO DOS DADOS
--------------------------------------------------------------------------------

CREATE OR REPLACE TRIGGER trg_empregados_biu_valida
BEFORE INSERT OR UPDATE ON tb_empregados
FOR EACH ROW
BEGIN
    -- Padronização dos campos textuais
    :NEW.nome  := UPPER(TRIM(:NEW.nome));
    :NEW.cargo := UPPER(TRIM(:NEW.cargo));

    -- Validação do salário mínimo definido no exercício
    IF :NEW.salario < 1412.00 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Violação: O salário informado não pode ser inferior ao piso da categoria.'
        );
    END IF;

    -- Impede redução salarial em operações de UPDATE
    IF UPDATING THEN
        IF :NEW.salario < :OLD.salario THEN
            RAISE_APPLICATION_ERROR(
                -20002,
                'Violação: Não é permitida a redução nominal do salário de um colaborador ativo.'
            );
        END IF;
    END IF;
END trg_empregados_biu_valida;
/


--------------------------------------------------------------------------------
-- ETAPA 3: TRIGGER AFTER ROW-LEVEL
-- TRILHA DE AUDITORIA
--------------------------------------------------------------------------------

CREATE OR REPLACE TRIGGER trg_empregados_auditoria
AFTER INSERT OR UPDATE OR DELETE ON tb_empregados
FOR EACH ROW
BEGIN
    IF INSERTING THEN

        INSERT INTO tb_auditoria_salarios (
            id_empregado,
            operacao,
            salario_anterior,
            salario_novo
        )
        VALUES (
            :NEW.id_empregado,
            'INSERT',
            NULL,
            :NEW.salario
        );

    ELSIF UPDATING THEN

        IF :NEW.salario <> :OLD.salario THEN
            INSERT INTO tb_auditoria_salarios (
                id_empregado,
                operacao,
                salario_anterior,
                salario_novo
            )
            VALUES (
                :OLD.id_empregado,
                'UPDATE',
                :OLD.salario,
                :NEW.salario
            );
        END IF;

    ELSIF DELETING THEN

        INSERT INTO tb_auditoria_salarios (
            id_empregado,
            operacao,
            salario_anterior,
            salario_novo
        )
        VALUES (
            :OLD.id_empregado,
            'DELETE',
            :OLD.salario,
            NULL
        );

    END IF;
END trg_empregados_auditoria;
/


--------------------------------------------------------------------------------
-- ETAPA 4: TRIGGER BEFORE STATEMENT-LEVEL
-- BLOQUEIO DE EXCLUSÕES NOS FINAIS DE SEMANA
--------------------------------------------------------------------------------

CREATE OR REPLACE TRIGGER trg_bloqueio_fim_semana
BEFORE DELETE ON tb_empregados
DECLARE
    v_dia_semana VARCHAR2(20);
BEGIN
    v_dia_semana := TO_CHAR(
        SYSDATE,
        'DY',
        'NLS_DATE_LANGUAGE=ENGLISH'
    );

    IF TRIM(v_dia_semana) IN ('SAT', 'SUN') THEN
        RAISE_APPLICATION_ERROR(
            -20005,
            'Operações de exclusão de dados são proibidas durante finais de semana.'
        );
    END IF;
END trg_bloqueio_fim_semana;
/

-- VIEW COM JOIN E REDIRECIONAMENTO DE INSERT

-- TABELA DE DEPARTAMENTOS

CREATE TABLE tb_departamentos (
    id_departamento NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome_departamento VARCHAR2(100) NOT NULL
);


-- TABELA DE FUNCIONÁRIOS DO DEPARTAMENTO

CREATE TABLE tb_funcionarios_depto (
    id_funcionario NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome_funcionario VARCHAR2(100) NOT NULL,
    cargo VARCHAR2(50) NOT NULL,
    id_departamento NUMBER NOT NULL,
    CONSTRAINT fk_func_depto
        FOREIGN KEY (id_departamento)
        REFERENCES tb_departamentos(id_departamento)
);


-- DADOS INICIAIS

INSERT INTO tb_departamentos (nome_departamento)
VALUES ('TECNOLOGIA');

INSERT INTO tb_departamentos (nome_departamento)
VALUES ('FINANCEIRO');

COMMIT;


-- VIEW COM JOIN

CREATE OR REPLACE VIEW vw_funcionarios_departamentos AS
SELECT
    f.id_funcionario,
    f.nome_funcionario,
    f.cargo,
    d.id_departamento,
    d.nome_departamento
FROM tb_funcionarios_depto f
JOIN tb_departamentos d
    ON d.id_departamento = f.id_departamento;


-- TRIGGER INSTEAD OF INSERT

CREATE OR REPLACE TRIGGER trg_vw_func_depto_insert
INSTEAD OF INSERT ON vw_funcionarios_departamentos
FOR EACH ROW
DECLARE
    v_id_departamento NUMBER;
BEGIN
    SELECT id_departamento
    INTO v_id_departamento
    FROM tb_departamentos
    WHERE UPPER(nome_departamento) = UPPER(TRIM(:NEW.nome_departamento));

    INSERT INTO tb_funcionarios_depto (
        nome_funcionario,
        cargo,
        id_departamento
    )
    VALUES (
        UPPER(TRIM(:NEW.nome_funcionario)),
        UPPER(TRIM(:NEW.cargo)),
        v_id_departamento
    );

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(
            -20010,
            'Departamento informado não encontrado.'
        );

    WHEN TOO_MANY_ROWS THEN
        RAISE_APPLICATION_ERROR(
            -20011,
            'Existe mais de um departamento com esse nome.'
        );
END trg_vw_func_depto_insert;
/

-- EXERCÍCIO 2: SIMULAÇÃO DE TABELA MUTANTE
-- ORA-04091
--------------------------------------------------------------------------------


-- TRIGGER PROPOSITALMENTE INCORRETO PARA DEMONSTRAÇÃO

CREATE OR REPLACE TRIGGER trg_empregados_mutante
AFTER INSERT OR UPDATE ON tb_empregados
FOR EACH ROW
DECLARE
    v_total NUMBER;
BEGIN
    SELECT COUNT(*)
    INTO v_total
    FROM tb_empregados
    WHERE cargo = :NEW.cargo;

    DBMS_OUTPUT.PUT_LINE(
        'Total de empregados no cargo: ' || v_total
    );
END trg_empregados_mutante;
/

-- EXERCÍCIO 3: CLÁUSULA WHEN
-- AUDITORIA SOMENTE QUANDO O SALÁRIO FOR ALTERADO
--------------------------------------------------------------------------------


-- SUBSTITUI O TRIGGER DE AUDITORIA ORIGINAL

CREATE OR REPLACE TRIGGER trg_empregados_auditoria
AFTER UPDATE OF salario ON tb_empregados
FOR EACH ROW
WHEN (NEW.salario <> OLD.salario)
BEGIN
    INSERT INTO tb_auditoria_salarios (
        id_empregado,
        operacao,
        salario_anterior,
        salario_novo
    )
    VALUES (
        :OLD.id_empregado,
        'UPDATE',
        :OLD.salario,
        :NEW.salario
    );
END trg_empregados_auditoria;
/
