/* TABELA DE PRODUTOS E ESTOQUE */

CREATE TABLE tb_prod_estoque (
    id_produto NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    descricao VARCHAR2(100) NOT NULL,
    preco NUMBER(10,2) NOT NULL,
    qtd_estoque NUMBER(6) NOT NULL
);


/* TABELA DE REGISTRO DE VENDAS */

CREATE TABLE tb_registro_vendas (
    id_venda NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_produto NUMBER REFERENCES tb_prod_estoque(id_produto),
    qtd_vendida NUMBER(6) NOT NULL,
    valor_venda NUMBER(12,2) NOT NULL,
    data_venda DATE DEFAULT SYSDATE NOT NULL
);

/* ============================================================
   ETAPA 2 - PACKAGE SPECIFICATION
   ESPECIFICAÇÃO DO PACOTE
   ============================================================ */

CREATE OR REPLACE PACKAGE pkg_gestao_vendas AS

    /* CONSTANTE PÚBLICA */
    c_taxa_desconto_padrao CONSTANT NUMBER := 0.05;

    /* EXCEÇÃO PÚBLICA */
    e_estoque_insuficiente EXCEPTION;

    /* PROCEDURE SOBRECARREGADA - VERSÃO 1 */
    PROCEDURE realizar_venda (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER,
        p_valor_final OUT NUMBER
    );

    /* PROCEDURE SOBRECARREGADA - VERSÃO 2 */
    PROCEDURE realizar_venda (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER,
        p_desconto_pct IN NUMBER,
        p_valor_final OUT NUMBER
    );

    /* FUNÇÃO PÚBLICA DE DISPONIBILIDADE */
    FUNCTION verificar_disponibilidade (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER
    ) RETURN BOOLEAN;

    /* FUNÇÃO PÚBLICA DE FATURAMENTO DIÁRIO */
    FUNCTION faturamento_diario
    RETURN NUMBER;

    /* EXERCÍCIO 1 - REPOSIÇÃO DE ESTOQUE */
    PROCEDURE repor_estoque (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER
    );

    /* EXERCÍCIO 3 - CONTADOR DE VENDAS DA SESSÃO */
    FUNCTION obter_total_vendas_sessao
    RETURN NUMBER;

END pkg_gestao_vendas;
/


/* ============================================================
   ETAPA 3 - PACKAGE BODY
   CORPO E IMPLEMENTAÇÃO DO PACOTE
   ============================================================ */

CREATE OR REPLACE PACKAGE BODY pkg_gestao_vendas AS

    /* VARIÁVEL PRIVADA DE SESSÃO */
    g_total_vendas_sessao NUMBER := 0;


    /* --------------------------------------------------------
       SUBPROGRAMAS PRIVADOS
       -------------------------------------------------------- */

    /* PROCEDURE PRIVADA PARA BAIXA DE ESTOQUE */

    PROCEDURE registrar_baixa_estoque (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER
    ) IS
    BEGIN
        UPDATE tb_prod_estoque
        SET qtd_estoque = qtd_estoque - p_quantidade
        WHERE id_produto = p_id_produto;
    END registrar_baixa_estoque;


    /* --------------------------------------------------------
       IMPLEMENTAÇÃO DA FUNÇÃO PÚBLICA
       VERIFICAR DISPONIBILIDADE
       -------------------------------------------------------- */

    FUNCTION verificar_disponibilidade (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER
    ) RETURN BOOLEAN IS

        v_estoque_atual NUMBER := 0;

    BEGIN
        IF p_quantidade IS NULL OR p_quantidade <= 0 THEN
            RETURN FALSE;
        END IF;

        SELECT qtd_estoque
        INTO v_estoque_atual
        FROM tb_prod_estoque
        WHERE id_produto = p_id_produto;

        RETURN (v_estoque_atual >= p_quantidade);

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RETURN FALSE;
    END verificar_disponibilidade;


    /* --------------------------------------------------------
       PROCEDURE SOBRECARREGADA - VERSÃO 1
       VENDA SEM DESCONTO MANUAL
       -------------------------------------------------------- */

    PROCEDURE realizar_venda (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER,
        p_valor_final OUT NUMBER
    ) IS
    BEGIN
        realizar_venda(
            p_id_produto,
            p_quantidade,
            0,
            p_valor_final
        );

    END realizar_venda;


    /* --------------------------------------------------------
       PROCEDURE SOBRECARREGADA - VERSÃO 2
       VENDA COM DESCONTO
       -------------------------------------------------------- */

    PROCEDURE realizar_venda (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER,
        p_desconto_pct IN NUMBER,
        p_valor_final OUT NUMBER
    ) IS

        v_preco_unitario NUMBER(10,2);
        v_total_bruto NUMBER(12,2);
        v_estoque_atual NUMBER(6);

    BEGIN

        /* VALIDAR QUANTIDADE */

        IF p_quantidade IS NULL OR p_quantidade <= 0 THEN
            RAISE_APPLICATION_ERROR(
                -20011,
                'A quantidade deve ser maior que zero.'
            );
        END IF;


        /* VALIDAR DESCONTO */

        IF p_desconto_pct IS NOT NULL
           AND (p_desconto_pct < 0 OR p_desconto_pct > 1) THEN

            RAISE_APPLICATION_ERROR(
                -20012,
                'O desconto deve estar entre 0 e 1.'
            );

        END IF;


        /* BLOQUEAR PRODUTO E CONSULTAR ESTOQUE */

        SELECT preco, qtd_estoque
        INTO v_preco_unitario, v_estoque_atual
        FROM tb_prod_estoque
        WHERE id_produto = p_id_produto
        FOR UPDATE;


        /* VALIDAR ESTOQUE */

        IF v_estoque_atual < p_quantidade THEN
            RAISE e_estoque_insuficiente;
        END IF;


        /* CALCULAR VALOR DA VENDA */

        v_total_bruto := v_preco_unitario * p_quantidade;

        p_valor_final :=
            v_total_bruto -
            (v_total_bruto * NVL(p_desconto_pct, 0));


        /* REALIZAR BAIXA DE ESTOQUE */

        registrar_baixa_estoque(
            p_id_produto,
            p_quantidade
        );


        /* REGISTRAR VENDA */

        INSERT INTO tb_registro_vendas (
            id_produto,
            qtd_vendida,
            valor_venda
        )
        VALUES (
            p_id_produto,
            p_quantidade,
            p_valor_final
        );


        /* ATUALIZAR CONTADOR DA SESSÃO */

        g_total_vendas_sessao :=
            g_total_vendas_sessao + 1;


        /* CONFIRMAR TRANSAÇÃO */

        COMMIT;


    EXCEPTION
        WHEN e_estoque_insuficiente THEN
            ROLLBACK;

            RAISE_APPLICATION_ERROR(
                -20010,
                'Estoque insuficiente para atender à quantidade solicitada.'
            );

        WHEN NO_DATA_FOUND THEN
            ROLLBACK;

            RAISE_APPLICATION_ERROR(
                -20013,
                'Produto não encontrado.'
            );

        WHEN OTHERS THEN
            ROLLBACK;

            RAISE_APPLICATION_ERROR(
                -20099,
                'Erro ao processar venda no pacote: ' || SQLERRM
            );

    END realizar_venda;


    /* --------------------------------------------------------
       FUNÇÃO PÚBLICA DE FATURAMENTO DIÁRIO
       -------------------------------------------------------- */

    FUNCTION faturamento_diario
    RETURN NUMBER IS

        v_faturamento NUMBER(12,2) := 0;

    BEGIN
        SELECT NVL(SUM(valor_venda), 0)
        INTO v_faturamento
        FROM tb_registro_vendas
        WHERE TRUNC(data_venda) = TRUNC(SYSDATE);

        RETURN v_faturamento;

    END faturamento_diario;


    /* --------------------------------------------------------
       EXERCÍCIO 1 - PROCEDURE PÚBLICA DE REPOSIÇÃO
       -------------------------------------------------------- */

    PROCEDURE repor_estoque (
        p_id_produto IN NUMBER,
        p_quantidade IN NUMBER
    ) IS

    BEGIN

        IF p_quantidade IS NULL OR p_quantidade <= 0 THEN
            RAISE_APPLICATION_ERROR(
                -20020,
                'A quantidade para reposição deve ser maior que zero.'
            );
        END IF;

        UPDATE tb_prod_estoque
        SET qtd_estoque = qtd_estoque + p_quantidade
        WHERE id_produto = p_id_produto;

        IF SQL%ROWCOUNT = 0 THEN
            RAISE_APPLICATION_ERROR(
                -20021,
                'Produto não encontrado para reposição.'
            );
        END IF;

        COMMIT;

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            RAISE;

    END repor_estoque;


    /* --------------------------------------------------------
       EXERCÍCIO 3 - FUNÇÃO DE CONTADOR DE VENDAS DA SESSÃO
       -------------------------------------------------------- */

    FUNCTION obter_total_vendas_sessao
    RETURN NUMBER IS

    BEGIN
        RETURN g_total_vendas_sessao;
    END obter_total_vendas_sessao;


    /* --------------------------------------------------------
       BLOCO DE INICIALIZAÇÃO DO PACOTE
       EXECUTADO UMA VEZ POR SESSÃO
       -------------------------------------------------------- */

BEGIN
    g_total_vendas_sessao := 0;

END pkg_gestao_vendas;
/
