/* ============================================================
   ETAPA 1 - CRIAÇÃO DO ESQUEMA BASE
   ============================================================ */

CREATE TABLE tb_contas (
    id_conta NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titular VARCHAR2(100) NOT NULL,
    saldo NUMBER(12,2) DEFAULT 0 NOT NULL,
    status VARCHAR2(10) DEFAULT 'ATIVA'
        CHECK (status IN ('ATIVA', 'BLOQUEADA'))
);

CREATE TABLE tb_movimentacoes (
    id_movimento NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_conta_origem NUMBER REFERENCES tb_contas(id_conta),
    id_conta_dest NUMBER REFERENCES tb_contas(id_conta),
    tipo_operacao VARCHAR2(20) NOT NULL,
    valor NUMBER(12,2) NOT NULL,
    data_movimento DATE DEFAULT SYSDATE NOT NULL
);

/* ============================================================
   ETAPA 2 - PROCEDURE COM PARÂMETROS IN E OUT
   DEPÓSITO VALIDADO
   ============================================================ */

CREATE OR REPLACE PROCEDURE prc_realizar_deposito (
    p_id_conta   IN NUMBER,
    p_valor      IN NUMBER,
    p_novo_saldo OUT NUMBER,
    p_status_msg OUT VARCHAR2
) IS
    v_status_conta VARCHAR2(10);
BEGIN
    IF p_valor IS NULL OR p_valor <= 0 THEN
        p_status_msg := 'ERRO: O valor do depósito deve ser maior que zero.';
        p_novo_saldo := NULL;
        RETURN;
    END IF;

    SELECT status, saldo
    INTO v_status_conta, p_novo_saldo
    FROM tb_contas
    WHERE id_conta = p_id_conta;

    IF v_status_conta <> 'ATIVA' THEN
        p_status_msg := 'ERRO: A conta está bloqueada ou inativa.';
        p_novo_saldo := NULL;
        RETURN;
    END IF;

    UPDATE tb_contas
    SET saldo = saldo + p_valor
    WHERE id_conta = p_id_conta
    RETURNING saldo INTO p_novo_saldo;

    INSERT INTO tb_movimentacoes (
        id_conta_dest,
        tipo_operacao,
        valor
    )
    VALUES (
        p_id_conta,
        'DEPOSITO',
        p_valor
    );

    COMMIT;

    p_status_msg := 'SUCESSO: Depósito efetuado com êxito.';

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        p_status_msg := 'ERRO: Conta inexistente.';
        p_novo_saldo := NULL;
        ROLLBACK;

    WHEN OTHERS THEN
        p_status_msg := 'ERRO INESPERADO: ' || SQLERRM;
        p_novo_saldo := NULL;
        ROLLBACK;
END prc_realizar_deposito;
/


/* ============================================================
   ETAPA 3 - EXECUÇÃO DA PROCEDURE DE DEPÓSITO
   ============================================================ */

DECLARE
    v_saldo_atual NUMBER;
    v_mensagem VARCHAR2(200);
BEGIN
    prc_realizar_deposito(
        p_id_conta   => 1,
        p_valor      => 1500.00,
        p_novo_saldo => v_saldo_atual,
        p_status_msg => v_mensagem
    );

    DBMS_OUTPUT.PUT_LINE('Resultado: ' || v_mensagem);
    DBMS_OUTPUT.PUT_LINE(
        'Novo Saldo: R$ ' ||
        TO_CHAR(v_saldo_atual, 'FM999G999G999D00')
    );
END;
/

/* ============================================================
   ETAPA 4 - PROCEDURE DE TRANSFERÊNCIA BANCÁRIA
   ============================================================ */

CREATE OR REPLACE PROCEDURE prc_transferir_fundos (
    p_origem  IN NUMBER,
    p_destino IN NUMBER,
    p_valor   IN NUMBER
) IS
    v_saldo_origem NUMBER;
    v_status_orig VARCHAR2(10);
    v_status_dest VARCHAR2(10);

    e_saldo_insuficiente EXCEPTION;
    e_conta_bloqueada EXCEPTION;
BEGIN
    IF p_valor IS NULL OR p_valor <= 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Valor de transferência inválido.'
        );
    END IF;

    IF p_origem = p_destino THEN
        RAISE_APPLICATION_ERROR(
            -20005,
            'A conta de origem e destino devem ser diferentes.'
        );
    END IF;

    SELECT saldo, status
    INTO v_saldo_origem, v_status_orig
    FROM tb_contas
    WHERE id_conta = p_origem
    FOR UPDATE;

    IF v_status_orig <> 'ATIVA' THEN
        RAISE e_conta_bloqueada;
    END IF;

    IF v_saldo_origem < p_valor THEN
        RAISE e_saldo_insuficiente;
    END IF;

    SELECT status
    INTO v_status_dest
    FROM tb_contas
    WHERE id_conta = p_destino
    FOR UPDATE;

    IF v_status_dest <> 'ATIVA' THEN
        RAISE e_conta_bloqueada;
    END IF;

    UPDATE tb_contas
    SET saldo = saldo - p_valor
    WHERE id_conta = p_origem;

    UPDATE tb_contas
    SET saldo = saldo + p_valor
    WHERE id_conta = p_destino;

    INSERT INTO tb_movimentacoes (
        id_conta_origem,
        id_conta_dest,
        tipo_operacao,
        valor
    )
    VALUES (
        p_origem,
        p_destino,
        'TRANSFERENCIA',
        p_valor
    );

    COMMIT;

    DBMS_OUTPUT.PUT_LINE(
        'Transferência de R$ ' ||
        TO_CHAR(p_valor, 'FM999G999G999D00') ||
        ' concluída com sucesso.'
    );

EXCEPTION
    WHEN e_saldo_insuficiente THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(
            -20002,
            'Falha: Saldo insuficiente na conta de origem.'
        );

    WHEN e_conta_bloqueada THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(
            -20003,
            'Falha: Uma das contas envolvidas está bloqueada.'
        );

    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(
            -20004,
            'Falha: Conta de origem ou destino não localizada.'
        );

    WHEN OTHERS THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(
            -20099,
            'Erro inesperado na transferência: ' || SQLERRM
        );
END prc_transferir_fundos;
/

/* ============================================================
   EXERCÍCIO 1 - PROCEDURE DE REAJUSTE DE SALDO EM MASSA
   ============================================================ */

CREATE OR REPLACE PROCEDURE prc_reajustar_saldo_em_massa (
    p_percentual IN NUMBER,
    p_total_atualizado OUT NUMBER
) IS
BEGIN
    IF p_percentual IS NULL OR p_percentual < 0 THEN
        RAISE_APPLICATION_ERROR(
            -20010,
            'O percentual deve ser maior ou igual a zero.'
        );
    END IF;

    UPDATE tb_contas
    SET saldo = saldo * (1 + p_percentual / 100)
    WHERE status = 'ATIVA';

    p_total_atualizado := SQL%ROWCOUNT;

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END prc_reajustar_saldo_em_massa;
/


/* TESTE DO REAJUSTE */

DECLARE
    v_total NUMBER;
BEGIN
    prc_reajustar_saldo_em_massa(

        p_total_atualizado => v_total
    );

    DBMS_OUTPUT.PUT_LINE(
        'Total de contas atualizadas: ' || v_total
    );
END;
/

/* ============================================================
   EXERCÍCIO 2 - PROCEDURE COM PARÂMETRO IN OUT
   SANITIZAÇÃO DE TEXTO
   ============================================================ */

CREATE OR REPLACE PROCEDURE prc_sanitizar_texto (
    p_texto IN OUT VARCHAR2
) IS
BEGIN
    p_texto := UPPER(TRIM(p_texto));
END prc_sanitizar_texto;
/

/* HISTÓRICO DE MOVIMENTAÇÕES */

SELECT
    m.id_movimento,
    m.id_conta_origem,
    origem.titular AS titular_origem,
    m.id_conta_dest,
    destino.titular AS titular_destino,
    m.tipo_operacao,
    m.valor,
    m.data_movimento
FROM tb_movimentacoes m
LEFT JOIN tb_contas origem
    ON origem.id_conta = m.id_conta_origem
LEFT JOIN tb_contas destino
    ON destino.id_conta = m.id_conta_dest
ORDER BY m.id_movimento;
