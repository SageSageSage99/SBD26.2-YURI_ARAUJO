/* TABELA DE COLABORADORES */

CREATE TABLE tb_colaboradores (
    id_colaborador NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    salario_base NUMBER(10,2) NOT NULL,
    departamento VARCHAR2(50) NOT NULL,

/* TABELA DE VENDAS DOS COLABORADORES */

CREATE TABLE tb_vendas_colaborador (
    id_venda NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_colaborador NUMBER REFERENCES tb_colaboradores(id_colaborador),
    mes_ano VARCHAR2(7) NOT NULL,
    total_vendido NUMBER(12,2) NOT NULL
);

/* ============================================================
   ETAPA 2 - FUNÇÃO DETERMINÍSTICA
   CÁLCULO DE IRRF
   ============================================================ */

CREATE OR REPLACE FUNCTION fnc_calcular_irrf (
    p_salario_bruto IN NUMBER
) RETURN NUMBER
DETERMINISTIC
IS
    v_imposto NUMBER(10,2) := 0;
BEGIN
    IF p_salario_bruto IS NULL OR p_salario_bruto <= 2259.20 THEN
        v_imposto := 0;

    ELSIF p_salario_bruto <= 2826.65 THEN
        v_imposto := (p_salario_bruto * 0.075) - 169.44;

    ELSIF p_salario_bruto <= 3751.05 THEN
        v_imposto := (p_salario_bruto * 0.15) - 381.44;

    ELSIF p_salario_bruto <= 4664.68 THEN
        v_imposto := (p_salario_bruto * 0.225) - 662.77;

    ELSE
        v_imposto := (p_salario_bruto * 0.275) - 896.00;
    END IF;

    RETURN ROUND(GREATEST(v_imposto, 0), 2);

EXCEPTION
    WHEN OTHERS THEN
        RETURN 0;
END fnc_calcular_irrf;
/


/* TESTE DA FUNÇÃO IRRF */

SELECT
    nome,
    salario_base,
    fnc_calcular_irrf(salario_base) AS imposto_irrf
FROM tb_colaboradores
ORDER BY nome;


/* ============================================================
   ETAPA 3 - FUNÇÃO COM CONSULTA AO BANCO
   CÁLCULO DE COMISSÃO DE VENDAS
   ============================================================ */

CREATE OR REPLACE FUNCTION fnc_obter_comissao (
    p_id_colaborador IN NUMBER,
    p_mes_ano IN VARCHAR2
) RETURN NUMBER
IS
    v_total_vendas NUMBER(12,2) := 0;
    v_percentual NUMBER(5,4) := 0;
BEGIN
    SELECT NVL(SUM(total_vendido), 0)
    INTO v_total_vendas
    FROM tb_vendas_colaborador
    WHERE id_colaborador = p_id_colaborador
      AND mes_ano = p_mes_ano;

    IF v_total_vendas >= 100000.00 THEN
        v_percentual := 0.05;

    ELSIF v_total_vendas >= 50000.00 THEN
        v_percentual := 0.035;

    ELSIF v_total_vendas > 0 THEN
        v_percentual := 0.02;

    ELSE
        v_percentual := 0;
    END IF;

    RETURN ROUND(v_total_vendas * v_percentual, 2);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN 0;

    WHEN OTHERS THEN
        RETURN 0;
END fnc_obter_comissao;
/

/* RELATÓRIO DE REMUNERAÇÃO */

SELECT
    c.id_colaborador,
    c.nome,
    c.departamento,
    c.salario_base,

    fnc_obter_comissao(
        c.id_colaborador,
        '10/2026'
    ) AS comissao_outubro,

    (
        c.salario_base +
        fnc_obter_comissao(c.id_colaborador, '10/2026')
    ) AS remuneracao_bruta,

    fnc_calcular_irrf(
        c.salario_base +
        fnc_obter_comissao(c.id_colaborador, '10/2026')
    ) AS desconto_irrf,

    (
        c.salario_base +
        fnc_obter_comissao(c.id_colaborador, '10/2026')
    ) -
    fnc_calcular_irrf(
        c.salario_base +
        fnc_obter_comissao(c.id_colaborador, '10/2026')
    ) AS salario_liquido_estimado

FROM tb_colaboradores c
ORDER BY remuneracao_bruta DESC;

/* ============================================================
   EXERCÍCIO 1 - FUNÇÃO DE TEMPO DE CASA
   ============================================================ */

CREATE OR REPLACE FUNCTION fnc_calcular_tempo_casa (
    p_data_admissao IN DATE
) RETURN NUMBER
IS
BEGIN
    IF p_data_admissao IS NULL THEN
        RETURN NULL;
    END IF;

    RETURN FLOOR(
        MONTHS_BETWEEN(SYSDATE, p_data_admissao) / 12
    );

END fnc_calcular_tempo_casa;
/

    data_admissao,
    fnc_calcular_tempo_casa(data_admissao) AS anos_de_empresa
FROM tb_colaboradores
ORDER BY nome;

    v.total_vendido

ORDER BY v.mes_ano, c.nome;


/* RESUMO DE COMISSÕES */

SELECT
    c.nome,
    v.mes_ano,
    SUM(v.total_vendido) AS total_vendido,
    fnc_obter_comissao(c.id_colaborador, v.mes_ano) AS comissao
FROM tb_colaboradores c
JOIN tb_vendas_colaborador v
    ON v.id_colaborador = c.id_colaborador
GROUP BY
    c.nome,
    c.id_colaborador,
    v.mes_ano
ORDER BY v.mes_ano, c.nome;
