    id_cliente NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    cidade VARCHAR2(50) NOT NULL,
    limite_credito NUMBER(12,2) NOT NULL,

/* ============================================================
   ETAPA 2 - CRIAÇÃO DO SYNONYM PRIVADO
   ============================================================ */

-- 1. Criar sinônimo privado apontando para a tabela base

CREATE OR REPLACE SYNONYM clientes
FOR tb_cad_clientes_corp_nacional;


-- 2. Consultar dados utilizando o sinônimo

SELECT
    id_cliente,
    codigo_empresa,
    razao_social,
    cidade,
    limite_credito
FROM clientes
WHERE status_conta = 'ATIVO';

/* ============================================================
   ETAPA 4 - ABSTRAÇÃO DE SEQUENCES COM SYNONYMS
   ============================================================ */

-- 1. Criar Sequence com nome técnico

CREATE SEQUENCE seq_tb_cad_clientes_corp_id
    START WITH 500
    INCREMENT BY 1
    NOCACHE;


-- 2. Criar sinônimo para a Sequence

CREATE OR REPLACE SYNONYM seq_clientes
FOR seq_tb_cad_clientes_corp_id;


-- 3. Consumir valores utilizando o sinônimo

SELECT
    seq_clientes.NEXTVAL AS proximo_id
FROM DUAL;


SELECT
    seq_clientes.CURRVAL AS id_atual
FROM DUAL;


/* ============================================================
   ETAPA 5 - REFATORAÇÃO TRANSPARENTE: SUBSTITUIÇÃO POR VIEW
   ============================================================ */

-- 1. Criar View que apresenta somente clientes ativos

CREATE OR REPLACE VIEW vw_clientes_ativos AS
SELECT
    id_cliente,
    codigo_empresa,
    razao_social,
    cidade,
    limite_credito
FROM tb_cad_clientes_corp_nacional
WHERE status_conta = 'ATIVO';


-- 2. Redirecionar o sinônimo para a View

CREATE OR REPLACE SYNONYM clientes
FOR vw_clientes_ativos;


-- 3. Consultar utilizando o mesmo sinônimo

SELECT *
FROM clientes;

/* ============================================================
   CONSULTA FINAL - IDENTIFICAÇÃO DE SINÔNIMOS ÓRFÃOS
   ============================================================ */

SELECT
    s.synonym_name,
    s.table_owner,
    s.table_name
FROM user_synonyms s
LEFT JOIN all_objects o
    ON o.owner = s.table_owner
   AND o.object_name = s.table_name
WHERE o.object_name IS NULL
ORDER BY s.synonym_name;
