/* ============================================================
   ETAPA 1 - CRIAÇÃO DE OBJECT TYPE COM ATRIBUTOS E MÉTODOS
   ============================================================ */

-- 1. Criar a especificação do tipo de objeto t_endereco

CREATE OR REPLACE TYPE t_endereco AS OBJECT (
    logradouro VARCHAR2(100),
    numero     VARCHAR2(10),
    bairro     VARCHAR2(50),
    cidade     VARCHAR2(50),
    uf         VARCHAR2(2),
    cep        VARCHAR2(9),

    MEMBER FUNCTION get_endereco_completo
    RETURN VARCHAR2
);
/

-- 2. Implementar o corpo do tipo t_endereco

CREATE OR REPLACE TYPE BODY t_endereco AS

    MEMBER FUNCTION get_endereco_completo
    RETURN VARCHAR2 IS
    BEGIN
        RETURN logradouro || ', ' ||
               numero || ' - ' ||
               bairro || ', ' ||
               cidade || '/' ||
               uf || ' (CEP: ' ||
               cep || ')';
    END;

END;
/


/* ============================================================
   ETAPA 2 - CRIAÇÃO DE COLEÇÕES (VARRAY E NESTED TABLE)
   ============================================================ */

-- 1. Criar VARRAY para telefones (máximo de 3 números)

CREATE OR REPLACE TYPE t_telefones_varray AS
VARRAY(3) OF VARCHAR2(20);
/

-- 2. Criar tipo de objeto para dependentes

CREATE OR REPLACE TYPE t_dependente AS OBJECT (
    nome_dependente VARCHAR2(80),
    grau_parentesco VARCHAR2(30),
    data_nascimento DATE
);
/

-- 3. Criar Nested Table baseada no objeto t_dependente

CREATE OR REPLACE TYPE t_dependentes_nested AS
TABLE OF t_dependente;
/


/* ============================================================
   ETAPA 3 - CRIAÇÃO DE TABELA COM TIPOS CUSTOMIZADOS
   ============================================================ */

CREATE TABLE tb_clientes_corporativos (
    id_cliente NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    razao_social VARCHAR2(100) NOT NULL,
    endereco t_endereco NOT NULL,
    telefones t_telefones_varray,
    dependentes t_dependentes_nested
)
NESTED TABLE dependentes
STORE AS tb_dependentes_store;

/* ============================================================
   EXERCÍCIO 1 - CRIAÇÃO DO TIPO t_retangulo
   ============================================================ */

-- 1. Criar a especificação do tipo

CREATE OR REPLACE TYPE t_retangulo AS OBJECT (
    largura NUMBER,
    altura NUMBER,

    MEMBER FUNCTION calcular_area
    RETURN NUMBER,

    MEMBER FUNCTION calcular_perimetro
    RETURN NUMBER
);
/

-- 2. Implementar o corpo do tipo

CREATE OR REPLACE TYPE BODY t_retangulo AS

    MEMBER FUNCTION calcular_area
    RETURN NUMBER IS
    BEGIN
        RETURN largura * altura;
    END;

    MEMBER FUNCTION calcular_perimetro
    RETURN NUMBER IS
    BEGIN
        RETURN 2 * (largura + altura);
    END;

END;
/


-- 3. Instanciar o objeto e testar os métodos

SELECT
    t_retangulo(10, 5).calcular_area() AS area,
    t_retangulo(10, 5).calcular_perimetro() AS perimetro
FROM DUAL;


/* ============================================================
   EXERCÍCIO 2 - MÉTODO MAP PARA ORDENAÇÃO
   ============================================================ */

-- 1. Atualizar a especificação do tipo t_retangulo

CREATE OR REPLACE TYPE t_retangulo AS OBJECT (
    largura NUMBER,
    altura NUMBER,

    MEMBER FUNCTION calcular_area
    RETURN NUMBER,

    MEMBER FUNCTION calcular_perimetro
    RETURN NUMBER,

    MAP MEMBER FUNCTION ordenar
    RETURN NUMBER
);
/

-- 2. Atualizar o corpo do tipo

CREATE OR REPLACE TYPE BODY t_retangulo AS

    MEMBER FUNCTION calcular_area
    RETURN NUMBER IS
    BEGIN
        RETURN largura * altura;
    END;

    MEMBER FUNCTION calcular_perimetro
    RETURN NUMBER IS
    BEGIN
        RETURN 2 * (largura + altura);
    END;

    MAP MEMBER FUNCTION ordenar
    RETURN NUMBER IS
    BEGIN
        RETURN largura * altura;
    END;

END;
/


-- 3. Criar tabela de retângulos

CREATE TABLE tb_retangulos (
    id_retangulo NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    retangulo t_retangulo
);
